cmake_minimum_required(VERSION 3.28)
project(AetherMindGoogleTestContract LANGUAGES C CXX)

set(CMAKE_CXX_STANDARD 20)
set(BUILD_TESTS ON CACHE BOOL "" FORCE)
set(BUILD_BENCHMARKS OFF CACHE BOOL "" FORCE)
set(BUILD_INFERENCE_TOOL OFF CACHE BOOL "" FORCE)
set(BUILD_DOCS OFF CACHE BOOL "" FORCE)
set(BUILD_WITH_TORCH OFF CACHE BOOL "" FORCE)
set(USE_LIBBACKTRACE OFF CACHE BOOL "" FORCE)
set(BACKTRACE_ON_SEGFAULT OFF CACHE BOOL "" FORCE)
set(AETHERMIND_SANITIZER none CACHE STRING "" FORCE)
set(AETHERMIND_VALIDATION_REQUIRED_ASSETS OFF CACHE BOOL "" FORCE)
# Any leftover find_package(GTest REQUIRED) must fail even on a machine with GTest.
set(CMAKE_DISABLE_FIND_PACKAGE_GTest ON)
set(FETCHCONTENT_FULLY_DISCONNECTED ON)
set(FETCHCONTENT_SOURCE_DIR_SPDLOG "${AETHERMIND_SPDLOG_SOURCE_DIR}" CACHE PATH "" FORCE)
set(FETCHCONTENT_SOURCE_DIR_GOOGLETEST "${AETHERMIND_GOOGLETEST_SOURCE_DIR}" CACHE PATH "" FORCE)

if(AETHERMIND_CONTRACT_CASE STREQUAL "disabled")
    set(BUILD_TESTS OFF CACHE BOOL "" FORCE)
elseif(AETHERMIND_CONTRACT_CASE STREQUAL "existing")
    set(BUILD_SHARED_LIBS ON)
    set(BUILD_GMOCK ON CACHE BOOL "" FORCE)
    set(INSTALL_GTEST ON CACHE BOOL "" FORCE)
    set(gtest_build_tests OFF CACHE BOOL "" FORCE)
    set(gtest_build_samples OFF CACHE BOOL "" FORCE)
    add_subdirectory("${AETHERMIND_GOOGLETEST_SOURCE_DIR}" host-googletest)
    foreach(target gtest gtest_main)
        foreach(property TYPE INTERFACE_COMPILE_DEFINITIONS INTERFACE_COMPILE_FEATURES MSVC_RUNTIME_LIBRARY)
            get_target_property(before_${target}_${property} GTest::${target} ${property})
            if("${before_${target}_${property}}" MATCHES "-NOTFOUND$")
                set(before_${target}_${property} "")
            endif()
        endforeach()
    endforeach()
elseif(AETHERMIND_CONTRACT_CASE STREQUAL "options")
    set(BUILD_GMOCK ON CACHE BOOL "" FORCE)
    set(INSTALL_GTEST ON CACHE BOOL "" FORCE)
elseif(AETHERMIND_CONTRACT_CASE MATCHES "^partial-(gtest|gtest_main)$")
    add_library(partial_gtest INTERFACE)
    add_library(GTest::${CMAKE_MATCH_1} ALIAS partial_gtest)
elseif(NOT AETHERMIND_CONTRACT_CASE STREQUAL "offline")
    message(FATAL_ERROR "Unknown GoogleTest contract case: ${AETHERMIND_CONTRACT_CASE}")
endif()

if(NOT AETHERMIND_CONTRACT_CASE MATCHES "^(offline|options)$")
    # A real fetch here must fail; these paths must never need GoogleTest sources.
    set(FETCHCONTENT_SOURCE_DIR_GOOGLETEST "${CMAKE_CURRENT_BINARY_DIR}/must-not-fetch-googletest" CACHE PATH "" FORCE)
endif()

add_subdirectory("${AETHERMIND_SOURCE_DIR}" aethermind)

if(AETHERMIND_CONTRACT_CASE STREQUAL "disabled")
    if(TARGET GTest::gtest OR TARGET GTest::gtest_main OR
       EXISTS "${CMAKE_CURRENT_BINARY_DIR}/_deps/googletest-subbuild" OR DEFINED BUILD_GMOCK)
        message(FATAL_ERROR "BUILD_TESTS=OFF must not prepare GoogleTest")
    endif()
    return()
endif()

if(AETHERMIND_CONTRACT_CASE STREQUAL "existing")
    foreach(target gtest gtest_main)
        foreach(property TYPE INTERFACE_COMPILE_DEFINITIONS INTERFACE_COMPILE_FEATURES MSVC_RUNTIME_LIBRARY)
            get_target_property(after_${target}_${property} GTest::${target} ${property})
            if("${after_${target}_${property}}" MATCHES "-NOTFOUND$")
                set(after_${target}_${property} "")
            endif()
            if(NOT "${before_${target}_${property}}" STREQUAL "${after_${target}_${property}}")
                message(FATAL_ERROR "AetherMind changed host GTest::${target} ${property}")
            endif()
        endforeach()
    endforeach()
endif()

if(AETHERMIND_CONTRACT_CASE MATCHES "^(existing|options)$")
    if(NOT BUILD_GMOCK OR NOT INSTALL_GTEST OR NOT TARGET GTest::gmock)
        message(FATAL_ERROR "AetherMind changed explicit GoogleTest options")
    endif()
elseif(BUILD_GMOCK OR INSTALL_GTEST OR gtest_build_tests OR gtest_build_samples OR TARGET GTest::gmock)
    message(FATAL_ERROR "Unexpected GoogleTest default options")
endif()

# Link only the public main target to exercise transitive headers and linkage.
add_executable(googletest_contract_probe
    "${AETHERMIND_SOURCE_DIR}/tests/cmake/test_required_execution_probe.cpp"
    "${AETHERMIND_SOURCE_DIR}/tests/unit/test_utils/test_required_execution_guard.cpp")
target_link_libraries(googletest_contract_probe PRIVATE GTest::gtest_main)
file(GENERATE OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/probe-location-$<CONFIG>.txt"
    CONTENT "$<TARGET_FILE:googletest_contract_probe>")
