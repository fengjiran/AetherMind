cmake_minimum_required(VERSION 3.28)
project(SharedDependencyContract NONE)

include(FetchContent)
include("${AETHERMIND_SOURCE_DIR}/cmake/AetherMindDependencySources.cmake")
if(CONTRACT_CASE STREQUAL "disabled")
    return()
elseif(CONTRACT_CASE STREQUAL "existing")
    add_subdirectory("${EXPECTED_SOURCE}" host-dependency)
elseif(CONTRACT_CASE STREQUAL "earlier")
    FetchContent_Declare(shared_probe SOURCE_DIR "${EXPECTED_SOURCE}")
elseif(CONTRACT_CASE STREQUAL "earlier-shared")
    aethermind_declare_shared_dependency(shared_probe "${EXPECTED_SOURCE}" "${TEST_REPOSITORY}" "${TEST_TAG}")
endif()
if(NOT TARGET shared_probe_target)
    aethermind_declare_shared_dependency(shared_probe "${SHARED_SOURCE}" "${TEST_REPOSITORY}" "${TEST_TAG}")
    aethermind_make_shared_dependency_available(shared_probe TARGETS shared_probe_target)
endif()
get_target_property(actual_source shared_probe_target SOURCE_DIR)
get_target_property(actual_binary shared_probe_target BINARY_DIR)
file(REAL_PATH "${actual_source}" actual_source)
file(REAL_PATH "${EXPECTED_SOURCE}" expected_source)
if(NOT actual_source STREQUAL expected_source)
    message(FATAL_ERROR "Wrong shared source: ${actual_source}; expected ${expected_source}")
endif()
cmake_path(IS_PREFIX CMAKE_BINARY_DIR "${actual_binary}" NORMALIZE binary_is_local)
if(NOT binary_is_local)
    message(FATAL_ERROR "Dependency binary directory escaped its build tree: ${actual_binary}")
endif()
if(EXISTS "${CMAKE_BINARY_DIR}/_deps/shared_probe-src")
    message(FATAL_ERROR "Shared source was copied into the build tree")
endif()
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/dependency-binary.txt" "${actual_binary}")
