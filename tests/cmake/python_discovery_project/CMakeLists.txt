cmake_minimum_required(VERSION 3.28)
project(AetherMindPythonDiscoveryContract LANGUAGES NONE)

include("${AETHERMIND_SOURCE_DIR}/cmake/AetherMindFindPython.cmake")

function(expect_python_found)
    if(NOT Python3_Interpreter_FOUND OR NOT AETHERMIND_PYTHON_FOUND OR
       NOT IS_ABSOLUTE "${Python3_EXECUTABLE}" OR NOT EXISTS "${Python3_EXECUTABLE}" OR
       Python3_VERSION VERSION_LESS 3.12 OR "${AETHERMIND_PYTHON_SOURCE}" STREQUAL "")
        message(FATAL_ERROR "Incomplete Python discovery result: ${Python3_Interpreter_FOUND}, ${Python3_EXECUTABLE}, ${Python3_VERSION}, ${AETHERMIND_PYTHON_SOURCE}")
    endif()
endfunction()

function(expect_python_disabled)
    if(Python3_Interpreter_FOUND OR AETHERMIND_PYTHON_FOUND)
        message(FATAL_ERROR "Disabled Python discovery retained a previous FOUND result")
    endif()
    foreach(output Python3_EXECUTABLE Python3_VERSION Python3_VERSION_MAJOR Python3_VERSION_MINOR
            Python3_VERSION_PATCH AETHERMIND_PYTHON_SOURCE)
        if(NOT "${${output}}" STREQUAL "")
            message(FATAL_ERROR "Disabled Python discovery retained ${output}=${${output}}")
        endif()
    endforeach()
endfunction()

if(AETHERMIND_PYTHON_CONTRACT_CASE STREQUAL "notfound")
    set(Python3_ARTIFACTS_INTERACTIVE TRUE)
    if(AETHERMIND_PYTHON_CONTRACT_STAGE STREQUAL "missing")
        # Exercise FindPython3's real failure cache without depending on which
        # Python versions happen to be installed on the host.
        file(MAKE_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/empty-root")
        set(CMAKE_FIND_ROOT_PATH "${CMAKE_CURRENT_BINARY_DIR}/empty-root")
        set(CMAKE_FIND_ROOT_PATH_MODE_PROGRAM ONLY)
        aethermind_find_python(MIN_VERSION 3.12)
        if(Python3_Interpreter_FOUND OR AETHERMIND_PYTHON_FOUND OR
           NOT "$CACHE{Python3_EXECUTABLE}" MATCHES "-NOTFOUND$")
            message(FATAL_ERROR "Missing Python did not produce the standard NOTFOUND cache")
        endif()
    elseif(AETHERMIND_PYTHON_CONTRACT_STAGE STREQUAL "recovered")
        if(NOT "$CACHE{Python3_EXECUTABLE}" MATCHES "-NOTFOUND$")
            message(FATAL_ERROR "Recovery must reuse the original NOTFOUND cache")
        endif()
        set(CMAKE_FIND_ROOT_PATH_MODE_PROGRAM NEVER)
        get_filename_component(python_dir "${AETHERMIND_TEST_PYTHON_EXECUTABLE}" DIRECTORY)
        set(CMAKE_PROGRAM_PATH "${python_dir}")
        aethermind_find_python(MIN_VERSION 3.12 REQUIRED)
        expect_python_found()
    else()
        message(FATAL_ERROR "Unknown NOTFOUND contract stage")
    endif()
elseif(AETHERMIND_PYTHON_CONTRACT_CASE MATCHES "^disabled(-required)?$")
    set(Python3_EXECUTABLE "${AETHERMIND_TEST_PYTHON_EXECUTABLE}")
    find_package(Python3 3.12 COMPONENTS Interpreter REQUIRED)
    if(NOT Python3_Interpreter_FOUND OR NOT EXISTS "${Python3_EXECUTABLE}")
        message(FATAL_ERROR "The parent must discover Python before disabling lookup")
    endif()
    # Seed every public result, as a previous helper call would do.
    set(AETHERMIND_PYTHON_FOUND TRUE)
    set(AETHERMIND_PYTHON_SOURCE system)
    set(CMAKE_DISABLE_FIND_PACKAGE_Python3 ON)
    if(AETHERMIND_PYTHON_CONTRACT_CASE STREQUAL "disabled-required")
        aethermind_find_python(MIN_VERSION 3.12 REQUIRED)
        message(FATAL_ERROR "REQUIRED Python discovery unexpectedly ignored the disable flag")
    endif()
    aethermind_find_python(MIN_VERSION 3.12)
    expect_python_disabled()
    enable_testing()
    if(Python3_Interpreter_FOUND)
        add_test(NAME unexpected_python COMMAND "${Python3_EXECUTABLE}" -c "print(42)")
    endif()
    get_property(registered_tests DIRECTORY PROPERTY TESTS)
    if(registered_tests)
        message(FATAL_ERROR "Disabled Python discovery registered a Python command")
    endif()
    set(CMAKE_DISABLE_FIND_PACKAGE_Python3 OFF)
    set(AETHERMIND_PYTHON_EXECUTABLE "${AETHERMIND_TEST_PYTHON_EXECUTABLE}" CACHE FILEPATH "")
    aethermind_find_python(MIN_VERSION 3.12 REQUIRED)
    expect_python_found()
elseif(AETHERMIND_PYTHON_CONTRACT_CASE STREQUAL "valid-pin")
    set(Python3_EXECUTABLE "${AETHERMIND_TEST_PYTHON_EXECUTABLE}" CACHE FILEPATH "")
    aethermind_find_python(MIN_VERSION 3.12 REQUIRED)
    expect_python_found()
    if(NOT "${Python3_EXECUTABLE}" STREQUAL "${AETHERMIND_TEST_PYTHON_EXECUTABLE}" OR
       NOT "$CACHE{Python3_EXECUTABLE}" STREQUAL "${AETHERMIND_TEST_PYTHON_EXECUTABLE}" OR
       NOT AETHERMIND_PYTHON_SOURCE STREQUAL "cached")
        message(FATAL_ERROR "The explicit Python pin was not preserved")
    endif()
elseif(AETHERMIND_PYTHON_CONTRACT_CASE STREQUAL "invalid-pin")
    if(AETHERMIND_PYTHON_CONTRACT_STAGE STREQUAL "relative")
        set(Python3_EXECUTABLE relative-python CACHE FILEPATH "" FORCE)
    else()
        set(Python3_EXECUTABLE "${CMAKE_CURRENT_BINARY_DIR}/missing-python" CACHE FILEPATH "" FORCE)
    endif()
    aethermind_find_python(MIN_VERSION 3.12)
    message(FATAL_ERROR "Invalid Python pin unexpectedly fell back to another interpreter")
else()
    message(FATAL_ERROR "Unknown Python discovery contract case")
endif()
